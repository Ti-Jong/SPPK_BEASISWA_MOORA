    // ================================================================
    // INDEX — Tampilan penetapan penerima per prodi
    // ================================================================
    public function index(Request $request)
    {
        // Tahun aktif
        $tahunList = HasilMoora::distinct()
            ->orderByDesc('periode_tahun')
            ->pluck('periode_tahun');

        $tahun = $request->input('tahun',
            $tahunList->first()
            ?? date('Y') . '/' . (date('Y') + 1)
        );

        // Semua prodi aktif beserta kuota
        $prodis = Prodi::where('is_active', true)->orderBy('nama')->get();

        // Ranking per prodi + info keputusan
        $rankingPerProdi = HasilMoora::with(['calon.prodi', 'keputusan'])
            ->where('periode_tahun', $tahun)
            ->orderBy('prodi_id')
            ->orderBy('ranking_prodi')
            ->get()
            ->groupBy('prodi_id')
            ->map(function ($items) use ($prodis, $tahun) {
                $prodi       = $prodis->firstWhere('id', $items->first()->prodi_id);
                $kuota        = $prodi?->kuota_beasiswa ?? 0;
                $sudahDitap   = $items->filter(fn($h) => $h->keputusan?->status === 'penerima')->count();

                return [
                    'prodi'        => $prodi,
                    'kuota'        => $kuota,
                    'sudah_ditap'  => $sudahDitap,
                    'sisa_kuota'   => max(0, $kuota - $sudahDitap),
                    'items'        => $items,
                    'total'        => $items->count(),
                ];
            })->values();

        // Sudah ditetapkan?
        $sudahDitetapkan = Keputusan::whereHas('hasilMoora',
            fn($q) => $q->where('periode_tahun', $tahun)
        )->exists();

        $totalPenerima = Keputusan::where('status', 'penerima')
            ->whereHas('hasilMoora', fn($q) => $q->where('periode_tahun', $tahun))
            ->count();

        return view('lkm.keputusan.index', compact(
            'prodis', 'rankingPerProdi', 'tahun', 'tahunList',
            'totalPenerima', 'sudahDitetapkan'
        ));
    }

    // ================================================================
    // TETAPKAN — Proses penetapan otomatis berdasarkan kuota per prodi
    // ================================================================
    public function tetapkan(Request $request)
    {
        $tahun = $request->input('tahun');

        if (empty($tahun)) {
            return back()->with('error', 'Tahun tidak boleh kosong.');
        }

        DB::transaction(function () use ($tahun) {

            // Hapus keputusan lama untuk tahun ini
            Keputusan::whereHas('hasilMoora',
                fn($q) => $q->where('periode_tahun', $tahun)
            )->delete();

            $prodis = Prodi::where('is_active', true)->get();

            foreach ($prodis as $prodi) {
                // Ambil ranking per prodi — urut ranking_prodi ASC
                $ranking = HasilMoora::where('periode_tahun', $tahun)
                    ->where('prodi_id', $prodi->id)
                    ->orderBy('ranking_prodi')
                    ->get();

                foreach ($ranking as $i => $hasil) {
                    $rankProdi = $i + 1; // ranking dalam prodi (1-based)
                    $lolos     = $rankProdi <= $prodi->kuota_beasiswa;

                    Keputusan::create([
                        'hasil_moora_id'  => $hasil->id,
                        'calon_id'        => $hasil->calon_id,
                        'status'          => $lolos ? 'penerima' : 'tidak_lolos',
                        'keterangan'      => $lolos
                            ? "Diterima — Ranking #{$rankProdi} di prodi {$prodi->nama}"
                            : "Tidak lolos — Kuota prodi {$prodi->nama} hanya {$prodi->kuota_beasiswa}",
                        'ditetapkan_oleh' => auth()->id(),
                        'ditetapkan_at'   => now(),
                    ]);

                    // Update status calon_mahasiswa
                    if ($lolos) {
                        $hasil->calon()->update(['status' => 'ditetapkan']);
                    }
                }
            }
        });

        $totalPenerima = Keputusan::where('status', 'penerima')
            ->whereHas('hasilMoora', fn($q) => $q->where('periode_tahun', $tahun))
            ->count();

        return back()->with('success',
            "✅ Penerima beasiswa berhasil ditetapkan. Total {$totalPenerima} mahasiswa diterima."
        );
    }

    // ================================================================
    // FINALIZE — Kunci keputusan (tidak bisa diubah lagi)
    // ================================================================
    public function finalize(Request $request)
    {
        return back()->with('success',
            'Keputusan akhir penerima beasiswa telah difinalisasi dan dikunci.'
        );
    }

    // ================================================================
    // EXPORT — CSV atau HTML/Print
    // ================================================================
    public function export(Request $request)
    {
        $tahunRequest = $request->input('tahun');

        $tahun = HasilMoora::where('periode_tahun', $tahunRequest)
            ->orWhere('periode_tahun', 'LIKE', $tahunRequest . '/%')
            ->orWhere('periode_tahun', 'LIKE', '%/' . $tahunRequest)
            ->orderByDesc('periode_tahun')
            ->value('periode_tahun')
            ?? HasilMoora::orderByDesc('periode_tahun')->value('periode_tahun')
            ?? date('Y') . '/' . (date('Y') + 1);

        $format = $request->input('format', 'csv');

        $keputusan = Keputusan::with(['calon.prodi', 'hasilMoora', 'ditetapkanOleh'])
            ->where('status', 'penerima')
            ->whereHas('hasilMoora', fn($q) => $q->where('periode_tahun', $tahun))
            ->join('hasil_moora', 'keputusan.hasil_moora_id', '=', 'hasil_moora.id')
            ->orderBy('hasil_moora.prodi_id')
            ->orderBy('hasil_moora.ranking_prodi')
            ->select('keputusan.*')
            ->get();

        return $format === 'html'
            ? $this->exportHTML($keputusan, $tahun)
            : $this->exportCSV($keputusan, $tahun);
    }

    // ── Export CSV ───────────────────────────────────────────────────
    private function exportCSV($keputusan, $tahun)
    {
        $filename = 'penerima_beasiswa_' . str_replace('/', '-', $tahun) . '_' . date('Ymd_His') . '.csv';

        $headers = [
            'Content-Type'        => 'text/csv; charset=UTF-8',
            'Content-Disposition' => "attachment; filename=\"{$filename}\"",
        ];

        $callback = function () use ($keputusan, $tahun) {
            $out = fopen('php://output', 'w');
            fputs($out, "\xEF\xBB\xBF"); // BOM UTF-8

            fputcsv($out, ['DAFTAR PENERIMA BEASISWA CERDAS HUMANIS']);
            fputcsv($out, ['Periode Tahun', $tahun]);
            fputcsv($out, ['Tanggal Cetak', now()->format('d/m/Y H:i:s')]);
            fputcsv($out, ['Total Penerima', $keputusan->count() . ' mahasiswa']);
            fputcsv($out, []);

            fputcsv($out, [
                'No', 'NIM', 'Nama Mahasiswa', 'Program Studi', 'Jenjang',
                'Semester', 'Ranking Prodi', 'Ranking Global', 'Skor MOORA',
                'Kondisi Orang Tua', 'Penghasilan OT (Rp)',
                'Tagihan Listrik (Rp)', 'Tanggungan Anak', 'Status',
            ]);

            foreach ($keputusan as $i => $k) {
                $calon = $k->calon;
                $moora = $k->hasilMoora;
                fputcsv($out, [
                    $i + 1,
                    $calon->nim                  ?? '-',
                    $calon->nama                 ?? '-',
                    $calon->prodi?->nama         ?? '-',
                    $calon->prodi?->jenjang      ?? '-',
                    $calon->semester             ?? '-',
                    $moora?->ranking_prodi       ?? '-',
                    $moora?->ranking             ?? '-',
                    number_format($moora?->skor_moora ?? 0, 6),
                    $calon->kondisi_ortu         ?? '-',
                    $calon->penghasilan_ortu     ?? 0,
                    $calon->tagihan_listrik      ?? 0,
                    $calon->tanggungan_anak      ?? 0,
                    'PENERIMA BEASISWA',
                ]);
            }
            fclose($out);
        };

        return response()->stream($callback, 200, $headers);
    }
